const API = "YOUR_APPS_SCRIPT_URL"; let nick = ""; let channel = "#general"; async function enterChat() { const n = document.getElementById("nickInput").value.trim(); if (!n) return; const res = await fetch(API, { method: "POST", body: JSON.stringify({ action: "checkNick", nick: n }) }); const data = await res.json(); if (data.exists) { alert("Nick taken"); return; } await fetch(API, { method: "POST", body: JSON.stringify({ action: "register", nick: n }) }); nick = n; document.getElementById("loginBox").style.display = "none"; load(); setInterval(load, 2500); // pseudo real-time IRC lag } // ========================= // IRC COMMANDS // ========================= async function sendMsg() { const input = document.getElementById("msgInput"); const text = input.value.trim(); if (!text) return; // /nick if (text.startsWith("/nick ")) { const newNick = text.split(" ")[1]; const res = await fetch(API, { method: "POST", body: JSON.stringify({ action: "nick", oldNick: nick, newNick }) }); const data = await res.json(); if (!data.ok) return alert(data.error); nick = newNick; input.value = ""; return; } // /join if (text.startsWith("/join ")) { channel = text.split(" ")[1]; await fetch(API, { method: "POST", body: JSON.stringify({ action: "join", nick, channel }) }); input.value = ""; load(); return; } // /who if (text === "/who") { const res = await fetch(API, { method: "POST", body: JSON.stringify({ action: "who", channel }) }); const data = await res.json(); alert("Users: " + data.users.join(", ")); input.value = ""; return; } // normal message await fetch(API, { method: "POST", body: JSON.stringify({ action: "send", nick, channel, message: text }) }); input.value = ""; load(); } // ========================= // LOAD MESSAGES // ========================= async function load() { const res = await fetch(API, { method: "POST", body: JSON.stringify({ action: "get", channel }) }); const data = await res.json(); const box = document.getElementById("messages"); box.innerHTML = ""; data.messages.forEach(m => { const div = document.createElement("div"); div.textContent = m; box.appendChild(div); }); box.scrollTop = box.scrollHeight; }
